<script>

    // Automatically fix landing page buttons to route to the app correctly

    document.addEventListener("DOMContentLoaded", function() {

        document.querySelectorAll("a, button").forEach(element => {

            let text = element.innerText.toLowerCase();

            let href = element.getAttribute("href") || "";

            

            if (text.includes("sign in") || text.includes("get started") || text.includes("launch web app") || href.includes("auth")) {

                element.setAttribute("href", "/login");

                if(element.tagName === "BUTTON") {

                    element.onclick = () => window.location.href = "/login";

                }

            }

        });

    });

</script>
